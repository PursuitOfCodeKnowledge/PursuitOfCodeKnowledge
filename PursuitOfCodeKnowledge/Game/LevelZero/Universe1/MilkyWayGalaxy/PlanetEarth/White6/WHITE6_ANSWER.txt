class Node:
	def __init__(self, value):
		self.value = value
		self.left = None
		self.right = None

def dfs(node):
	if node is not None:
		print(node.value) # Обработка узла
		dfs(node.left) # Рекурсивный обход левого поддерева
		dfs(node.right) # Рекурсивный обход правого поддерева

def dfs_with_nodes(node, parent_value = None):
	if node is not None:
		"""Печатаем значение узла и его родительский узел, если он есть"""
		if parent_value is not None:
			print(f"{parent_value} -> {node.value}")
		else:
			print(f"root -> {node.value}") # Корень дерева

		dfs_with_nodes(node.left, node.value) # Рекурсивный обход левого поддерева
		dfs_with_nodes(node.right, node.value) # Рекурсивный обход правого поддерева

def print_tree(node, level = 0):
	if node is not None:
		print_tree(node.right, level + 1) # Сначала печатаем правое поддерево
		print(' ' * 4 * level + f"-> {node.value}") # Печатаем текущий узел с отступами
		print_tree(node.left, level + 1) # Затем печатаем левое поддерево

if __name__ == "__main__":
	root = Node(50)
	root.left = Node(40)
	root.right = Node(60)
	root.left.left = Node(30)
	root.left.right = Node(45)
	root.right.left = Node(55)
	root.right.right = Node(65)
	root.left.left.left = Node(25)
	root.left.left.right = Node(35)
	root.left.right.left = Node(42)
	root.left.right.right = Node(47)
	root.right.right.left = Node(62)
	root.right.right.right = Node(67)
	print(f"root = Node(50)\nroot.left = Node(40)\nroot.right = Node(60)\nroot.left.left = Node(30)\nroot.left.right = Node(45)\nroot.right.left = Node(55)\nroot.right.right = Node(65)\nroot.left.left.left = Node(25)\nroot.left.left.right = Node(35)\nroot.left.right.left = Node(42)\nroot.left.right.right = Node(47)\nroot.right.right.left = Node(62)\nroot.right.right.right = Node(67)")
	print_tree(root)
	print("DFS обход дерева в глубину: ")
	dfs(root)
	dfs_with_nodes(root)

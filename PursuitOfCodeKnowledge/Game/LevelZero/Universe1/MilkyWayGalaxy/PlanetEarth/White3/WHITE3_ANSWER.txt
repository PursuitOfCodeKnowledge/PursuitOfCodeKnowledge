from single_node import SingleNode;
class SingleLinkedList:
  def __init__(self):
    self.head = None

  def append(self, data):
    """ Добавляет узел в конец списка """
    new_node = SingleNode(data)
    if not self.head: # Если список пуст
      self.head = new_node
      return
    last_node = self.head
    while last_node.next: # Проходим до последнего узла
      last_node = last_node.next
    last_node.next = new_node # Указываем на новый узел

  def print_list_single(self):
    """Выводит элементы односвязного списка """
    current_node = self.head
    while current_node: # Перебираем узлы
      print(f"{current_node.data}", end=' ')
      current_node = current_node.next
    print()

  def delete_node(self, key):
    """Удаляет узел с заданным значением"""
    current_node = self.head
    if current_node is not None:
      if current_node.data == key: # Если мы хотим удалить первую запись
        self.head = current_node.next # Смена головы на следующий узел
        current_node = None # Удаляем текущий узел
        return

    # Проходим по списку
    while current_node is not None:
      if current_node.data == key: # Найден узел для удаления
        break
      prev_node = current_node
      current_node = current_node.next

    if current_node is None: # Если узел не найден
      return

    prev_node.next = current_node.next # Соедидняем предыдущий узел с следующим
    current_node = None # Удаляем узел

  def reverse_single_node(self):
    """Обратный порядок в односвязном списке"""
    prev_node = None
    current_node = self.head
    while current_node is not None:
      next_node = current_node.next # Сохраняем следующий узел
      current_node.next = prev_node # Поворачиваем стрелку
      prev_node = current_node # Сдвигаем prevNode на следующий узел
      current_node = next_node # Переходим к следующему узлу
    self.head = prev_node # Обновляем голову на последний узел


if __name__ == "__main__":
  single_linked_list = SingleLinkedList()
  for i in range(10):
    single_linked_list.append(i)
  print("Исходный список: ")
  single_linked_list.print_list_single()
  print("В обратном порядке односвязный список: ")
  single_linked_list.reverse_single_node()
  single_linked_list.print_list_single()
  for i in range(8):
    single_linked_list.delete_node(i)
  print("Исходный список: ")
  single_linked_list.print_list_single()

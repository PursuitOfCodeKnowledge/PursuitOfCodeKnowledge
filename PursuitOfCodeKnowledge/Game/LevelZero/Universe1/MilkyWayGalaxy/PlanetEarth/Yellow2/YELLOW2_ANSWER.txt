import networkx as nx
import matplotlib.pyplot as plt

def undirected_graph():
	# Создаем неориентированный граф
	graph = nx.Graph()
	# Добавление узлов
	graph.add_nodes_from([1, 2, 3, 4])
	# Добавление рёбер
	graph.add_edges_from([(1, 2), (1, 3), (2, 4), (3, 4)])
	return graph

def print_graph(graph):
	print("Граф:")
	for node in graph.nodes():
		# Получаем соседей узла
		neighbors = list(graph.neighbors(node))
		if neighbors:
			print(f"{node} -> {', '.join(map(str, neighbors))}")

def undirected_graph_visualization(graph):
	# Визуализация графа
	nx.draw(graph, with_labels=True, node_color='lightblue', node_size=700, font_size=16)
	plt.title("Неориентированный граф")
	plt.show()

if __name__ == "__main__":
	# Печатаем граф неориентированный
	graph = undirected_graph()
	print_graph(graph)
	undirected_graph_visualization(graph)

--1) Функция принимает ticket_no и возвращает итоговую сумму с применением скидки 10% для Business (используй IF и RAISE NOTICE для отладки).
-- Создать функцию
CREATE OR REPLACE FUNCTION bookings.get_adjusted_fare(p_ticket_no CHAR(13))
RETURNS NUMERIC(10,2) LANGUAGE plpgsql AS $$
DECLARE
    v_fare NUMERIC(10,2);
    v_fare_cond TEXT;
    v_result NUMERIC(10,2);
BEGIN
    SELECT amount, fare_conditions
    INTO v_fare, v_fare_cond
    FROM bookings.ticket_flights
    WHERE ticket_no = p_ticket_no
    LIMIT 1;

    IF v_fare IS NULL THEN
        RAISE NOTICE 'Ticket % not found in ticket_flights', p_ticket_no;
        RETURN NULL;
    END IF;

    -- Если Business -> скидка 10%
    IF v_fare_cond = 'Business' THEN
        v_result := ROUND(v_fare * 0.90, 2);  -- 10% off
        RAISE NOTICE 'Business fare: original=%, discounted=%', v_fare, v_result;
    ELSE
        v_result := v_fare;
        RAISE NOTICE 'No discount for fare_condition=%', v_fare_cond;
    END IF;

    RETURN v_result;
END;
$$;

--2) Функция возвращает категорию тарифа для конкретной записи ticket_flights (используй CASE).
CREATE OR REPLACE FUNCTION bookings.fare_category(p_ticket_no CHAR(13))
RETURNS TEXT LANGUAGE plpgsql AS $$
DECLARE
    v_amount NUMERIC(10,2);
    v_cat TEXT;
BEGIN
    SELECT amount INTO v_amount
    FROM bookings.ticket_flights
    WHERE ticket_no = p_ticket_no
    LIMIT 1;

    IF v_amount IS NULL THEN
        RETURN 'UNKNOWN';

    v_cat := CASE

        WHEN v_amount >= 50000 THEN 'Medium'
        ELSE 'Low'
    END;

    RETURN v_cat;
END;
$$;

Пример 3: 
--3) Процедура: пройти по всем билетам бронирования и вывести пассажиров + суммировать общую сумму(используй FOR ... IN ... LOOP, RAISE NOTICE).
CREATE OR REPLACE PROCEDURE bookings.show_passengers_and_sum(p_book_ref CHAR(6))
LANGUAGE plpgsql AS $$
DECLARE
    rec RECORD;
    v_total NUMERIC(12,2) := 0;
BEGIN
    FOR rec IN
        SELECT t.ticket_no, t.passenger_name, COALESCE(SUM(tf.amount),0) AS ticket_total
        FROM bookings.tickets t
        LEFT JOIN bookings.ticket_flights tf ON tf.ticket_no = t.ticket_no
        WHERE t.book_ref = p_book_ref
        GROUP BY t.ticket_no, t.passenger_name
    LOOP
        RAISE NOTICE 'Ticket=% Passenger=% TicketTotal=%', rec.ticket_no, rec.passenger_name, rec.ticket_total;
        v_total := v_total + rec.ticket_total;
    END LOOP;

    RAISE NOTICE 'Booking % total computed = %', p_book_ref, v_total;
END;
$$;

--4) Функция обойти массив flight_id для заданного ticket_no и обновить локальную переменную(используй while).
CREATE OR REPLACE FUNCTION bookings.process_flights_by_ticket(p_ticket_no CHAR(13))
RETURNS INTEGER LANGUAGE plpgsql AS $$
DECLARE
    v_flights INT[] := ARRAY(SELECT flight_id FROM bookings.ticket_flights WHERE ticket_no = p_ticket_no ORDER BY flight_id);
    i INT := 1;
    v_count INT := COALESCE(array_length(v_flights,1), 0);
    v_fid INT;
BEGIN
    IF v_count = 0 THEN
        RAISE NOTICE 'No flights for ticket %', p_ticket_no;
        RETURN 0;
    END IF;

    WHILE i <= v_count LOOP
        v_fid := v_flights[i];
        -- пример действия: просто печатаем и, например, помечаем (логика для примера)
        RAISE NOTICE 'Processing flight #% (flight_id=%)', i, v_fid;
        i := i + 1;
    END LOOP;

    RETURN v_count;
END;
$$;

Пример 5: 
--5) Процедура: для каждого билета в бронировании проверяет, есть ли flights(используй IF, CASE, LOOP и WHILE вместе).
CREATE OR REPLACE PROCEDURE bookings.audit_booking(p_book_ref CHAR(6))
LANGUAGE plpgsql AS $$
DECLARE
    t_rec RECORD;
    flight_ids INT[];
    idx INT;
    cnt INT;
    ticket_sum NUMERIC(12,2);
    cat TEXT;
BEGIN
    FOR t_rec IN
        SELECT ticket_no FROM bookings.tickets WHERE book_ref = p_book_ref
    LOOP
        -- собрать flight_id для билета
        SELECT ARRAY_AGG(flight_id ORDER BY flight_id) INTO flight_ids
        FROM bookings.ticket_flights
        WHERE ticket_no = t_rec.ticket_no;

        IF flight_ids IS NULL THEN
            RAISE NOTICE 'Ticket % has no flights', t_rec.ticket_no;
            CONTINUE;
        END IF;

        cnt := COALESCE(array_length(flight_ids,1), 0);
        idx := 1;
        ticket_sum := 0;

        -- WHILE по flight_ids
        WHILE idx <= cnt LOOP
            SELECT amount INTO STRICT ticket_sum
            FROM bookings.ticket_flights
            WHERE ticket_no = t_rec.ticket_no AND flight_id = flight_ids[idx];

            -- Классифицируем
            cat := CASE
                WHEN ticket_sum >= 100000 THEN 'High'
                WHEN ticket_sum >= 50000 THEN 'Medium'
                ELSE 'Low'
            END;

            RAISE NOTICE 'Ticket % Flight %: amount=% category=%', t_rec.ticket_no, flight_ids[idx], ticket_sum, cat;

            idx := idx + 1;
        END LOOP;
    END LOOP;
END;
$$;

import networkx as nx
import matplotlib.pyplot as plt

class WeightedGraph:
	def __init__(self):
		self.graph = nx.Graph()

	def create_graph(self, edges):
		"""Создание графа и добавление рёбер с весами """
		self.graph.add_weighted_edges_from(edges)

	def find_shortest_graph(self, source, target):
		"""Поиск кратчайшего пути между двумя узлами"""
		try:
			return nx.shortest_path(self.graph, source=source, target=target, weight='weight')
		except nx.NetworkXNoPath:
			print(f"Нет пути между узлами {source} и {target}.")
			return None
		except nx.NodeNotFound as e:
			print(f"Ошибка: {e}. Проверьте существуют ли узлы {source} и {target}.")
			return None

	def visualize_graph(self):
		""" Визуализация графа с помощью matplotlib."""
		if self.graph.number_of_nodes() == 0:
			print("Граф пуст. Не возможно визуализировать.")
			return

		pos = nx.spring_layout(self.graph)
		nx.draw(self.graph, pos, with_labels=True, node_color='lightyellow', node_size=700, font_size=16)
		edge_labels = nx.get_edge_attributes(self.graph, 'weight')
		nx.draw_networkx_edge_labels(self.graph, pos, edge_labels=edge_labels)
		plt.title("Взешенный граф:")
		plt.show()

	def print_graph(self):
		"""Вывод графа на терминал """
		if self.graph.number_of_edges() == 0:
			print("Граф пуст. Невозможно вывести")
			return

		print("Граф с весами: ")
		for (u, v, weight) in self.graph.edges(data='weight'):
			print(f"Узел {u} -- Узел {v} (Вес: {weight})")

if __name__ == "__main__":
	edges = [(1, 2, 1), (1, 3, 4), (2, 3, 2), (2, 4, 5), (3, 4, 1)]
	graph = WeightedGraph()
	graph.create_graph(edges)
	print("Кратчайший путь от 1 до 4: ", graph.find_shortest_graph(1, 4))
	graph.print_graph()
	graph.visualize_graph()

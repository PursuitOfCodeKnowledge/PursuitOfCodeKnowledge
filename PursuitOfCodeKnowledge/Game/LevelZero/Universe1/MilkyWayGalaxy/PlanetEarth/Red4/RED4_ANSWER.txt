def binary_search_agi(arr, target):
  """
    Функция для бинарного поиска элемента в отсортированном массиве.

    :param arr: Список в котором производится поиск
    :param target: Искомый элемент
    :return: Индекс искомого элемента или -1, если элемент не найден
  """
  left, right = 0, len(arr) - 1
  while left <= right:
    mid = left + (right-left) // 2
    if arr[mid] == target:
      return mid
    elif arr[mid] < target:
      left = mid + 1
    else:
      right = mid - 1
  return -1

def main():
    my_list_binary_search = [x for x in range(99)]
    my_list_binary_search.sort()
    print(my_list_binary_search)
    try:
      my_target = int(input("Введите целое число: "))
      result = binary_search_agi(my_list_binary_search, my_target)

      if result != -1:
        print(f"Элемент {my_target} найден в списке: {result}")
      else:
        print(f"Элемент {my_target} не найден в списке")
    except ValueError:
      print("Пожалуйста введите корректное целое число")

if __name__ == "__main__":
    main()

import networkx as nx
import matplotlib.pyplot as plt

def directed_graph():
	# Создание ориентированного графа
	graph = nx.DiGraph()
	# Добавление узлов
	graph.add_nodes_from([1, 2, 3, 4])
	# Добавление рёбер с направлениями
	graph.add_edges_from([(1, 2), (2, 3), (3, 1), (3, 4)])
	return graph

def print_directed_graph(graph):
	# Выводим граф в терминале
	print("Граф ориентированный:")
	for node in graph.nodes():
		edges = list(graph.successors(node))
		if edges:
			print(f"{node} -> {', '.join(map(str, edges))}")
		else:
			print(f"{node} -> (нет выходящих рёбер)")

def directed_graph_visualization(graph):
	# Визуализация ориентированного графа
	nx.draw(graph, with_labels=True, node_color='lightgreen', node_size=700, font_size=16, arrows=True)
	plt.title("Ориентированный граф:")
	plt.show()

if __name__ == "__main__":
	graph = directed_graph()
	print_directed_graph(graph) # Вывод графа в терминале
	directed_graph_visualization(graph) # Визуализация графа

from double_node import DoubleNode
class DoubleLinkedList:
	def __init__(self):
		self.head = None

	def append(self, data):
		"""Добавление узла в конец списка """
		new_node = DoubleNode(data);
		if not self.head: # Если список пуст
			self.head = new_node
			return
		last = self.head
		while last.next:
			last = last.next
		last.next = new_node
		new_node.prev = last

	def prepend(self, data):
		"""Добавление узла в начало списка."""
		new_node = DoubleNode(data)
		if not self.head: # Если список пуст
			self.head = new_node
			return
		new_node.next = self.head
		self.head.prev = new_node
		self.head = new_node

	def delete_node(self, key):
		"""Удаление узла по значению."""
		current = self.head
		while current:
			if current.data == key: # Узел найден
				if current.prev: # Если это не первый узел
					current.prev.next = current.next
				if current.next: # Если это не последний узел
					current.next.prev = current.prev
				if current == self.head: # Если это голова списка
					self.head = current.next
				current = None
				return
			current = current.next

	def reverse(self):
		"""Обратный порядок в двусвязном списке"""
		current = self.head
		temp = None
		# Меняем местами ссылки next и prev для каждого узла
		while current:
			temp = current.prev # Храним предыдущий узел
			current.prev = current.next # Меняем next на prev
			current.next = temp # Меняем prev на  next
			current = current.prev # Переходим к следующему узлу(который теперь предшествующий)
		if temp: # Если список не пустой, обновляем голову
			self.head = temp.prev

	def search(self, value):
		"""Ищет узел по значению и возвращает его, если найден, иначе - None. """
		current = self.head
		while current:
			if current.data == value:
				return current # Возвращает узел, если значение найдено
			current = current.next
		return None # Возвращает None, если значение не найдено

	def print_reverse(self):
		"""Печать всех элементов списка от конца до головы."""
		current = self.head
		while current and current.next: # Найти последний узел
			current = current.next
		while current: # Печатаем в обратном порядке
			print(f"{current.data}", end = ' ')
			current = current.prev
		print() # Для переноса строки

	def print_list(self):
		"""Печать всех элементов списка от головы до конца"""
		current = self.head
		while current:
			print(f"{current.data}", end= ' ')
			current = current.next
		print() # Для переноса строки

from double_node import DoubleNode
class LinkedListDouble:
	def __init__(self):
		self.head = None

	def append(self, data):
		"""Добавление узла в конец списка """
		new_node = DoubleNode(data)
		if self.head is None: # Если список пуст
			self.head = new_node
			return
		last = self.head
		while last.next: # Найти последний узел
			last = last.next
		last.next = new_node # Установить следующий до последнего узла
		new_node.prev = last # Установить предыдущий для нового узла

	def prepend(self, data):
		"""Добавление узла в начало списка."""
		new_node = DoubleNode(data)
		if self.head is None: # Если список пуст
			self.head = new_node
			return
		new_node.next = self.head # Установить следующий для нового узла
		self.head.prev = new_node # Установить предыдущий для старого узла
		self.head = new_node # Обновить голову списка

	def delete_node(self, key):
		"""Удаление узла по значению."""
		current = self.head
		if current is not None:
			if current.data == key: # Если узел, который нужно удалить, - это голова
				self.head = current.next # Обновление головы
				if self.head: # Если следующий узел существует
					self.head.prev = None
				current = None # Удаляем узел
				return

		while current is not None:
			if current.data == key: # Узел найден для удаления
				if current.next: # Если есть следующий
					current.next.prev = current.prev
				if current.prev: # Если есть предыдущий
					current.prev.next = current.next
				current = None # Удаляем узел
				return

			current = current.next # Переходим к следующему узлу

	def insert_at_position(self, data, position):
		"""Вставляе узел с данным значением в указанную позицию(индексация начинается с нуля)."""
		if position < 0:
			print("Позиция должна быть целым не отрицательным числом")
			return
		if position == 0:
			self.prepend(data) # Вставка в начало
			return

		new_node = DoubleNode(data)
		current = self.head
		index = 0

		# Перемещение к необходимой позиции
		while current and index < position:
			current = current.next
			index += 1

		if current is None: # Вставка в конец списка
			self.append(data)
		else: # Вставка перед текущим узлом
			new_node.prev = current.prev
			new_node.next = current
			if current.prev:
				current.prev.next = new_node
			current.prev = new_node

	def remove_at_position(self, position):
		"""Удаляет из произвольной позиции двусвязного списка"""
		if position < 0:
			print("Ошибка: позиция не может быть отрицательной")
			return

		current = self.head
		index = 0

		# Перемещение к необходимой позиции
		while current and index < position:
			current = current.next
			index += 1

		# Если текущий узел - это None, позиция вне границ
		if current is None:
			print("Ошибка: позиция вне границ списка.")
			return

		# Удаление узла
		if current.prev:
			current.prev.next = current.next # Устанавливаем ссылку на следующий элемент
		if current.next:
			current.next.prev = current.prev # Устанавливаем ссылку на предыдущий узел
		if current == self.head:
			self.head = current.next # Если это голова, перемещаем голову на следующий узел

		current = None # Собираем мусор, отключая узел

	def reverse(self):
		"""Обратный порядок в двусвязном списке"""
		current = self.head
		temp = None
		# Меняем местами ссылки next и prev для каждого узла
		while current:
			temp = current.prev # Храним предыдущий узел
			current.prev = current.next # Меняем next на prev
			current.next = temp # Меняем prev на  next
			current = current.prev # Переходим к следующему узлу(который теперь предшествующий)
		if temp: # Если список не пустой, обновляем голову
			self.head = temp.prev

	def search(self, value):
		"""Ищет узел по значению и возвращает его, если найден, иначе - None. """
		current = self.head
		while current:
			if current.data == value:
				return current # Возвращает узел, если значение найдено
			current = current.next
		return None # Возвращает None, если значение не найдено

	def search_return_index(self, value):
		"""Ищет узел по значению и возвращает его индекс, если найден, иначе возвращает -1"""
		current = self.head
		index = 0
		while current:
			if current.data == value:
				return index
			current = current.next
			index += 1
		return -1 # Если элемент не найден

	def print_reverse(self):
		"""Печать всех элементов списка от конца до головы."""
		current = self.head
		while current and current.next: # Найти последний узел
			current = current.next
		while current: # Печатаем в обратном порядке
			print(f"{current.data}", end = ' <-> ')
			current = current.prev
		print("None") # Для переноса строки

		current = self.head
		while current:
			print(f"{current.data}", end= ' <-> ')
			current = current.next

		"""Добавление узла в начало списка."""
		new_node = DoubleNode(data)
		if self.head is None: # Если список пуст
			self.head = new_node
			return
		new_node.next = self.head # Установить следующий для нового узла
		self.head.prev = new_node # Установить предыдущий для старого узла
		self.head = new_node # Обновить голову списка

	def delete_node(self, key):
		"""Удаление узла по значению."""
		current = self.head
		if current is not None:
			if current.data == key: # Если узел, который нужно удалить, - это голова
				self.head = current.next # Обновление головы
				if self.head: # Если следующий узел существует
					self.head.prev = None
				current = None # Удаляем узел
				return

		while current is not None:
			if current.data == key: # Узел найден для удаления
				if current.next: # Если есть следующий
					current.next.prev = current.prev
				if current.prev: # Если есть предыдущий
					current.prev.next = current.next
				current = None # Удаляем узел
				return

			current = current.next # Переходим к следующему узлу

	def insert_at_position(self, data, position):
		"""Вставляе узел с данным значением в указанную позицию(индексация начинается с нуля)."""
		if position < 0:
			print("Позиция должна быть целым не отрицательным числом")
			return
		if position == 0:
			self.prepend(data) # Вставка в начало
			return

		new_node = DoubleNode(data)
		current = self.head
		index = 0

		# Перемещение к необходимой позиции
		while current and index < position:
			current = current.next
			index += 1

		if current is None: # Вставка в конец списка
			self.append(data)
		else: # Вставка перед текущим узлом
			new_node.prev = current.prev
			new_node.next = current
			if current.prev:
				current.prev.next = new_node
			current.prev = new_node

	def remove_at_position(self, position):
		"""Удаляет из произвольной позиции двусвязного списка"""
		if position < 0:
			print("Ошибка: позиция не может быть отрицательной")
			return

		current = self.head
		index = 0

		# Перемещение к необходимой позиции
		while current and index < position:
			current = current.next
			index += 1

		# Если текущий узел - это None, позиция вне границ
		if current is None:
			print("Ошибка: позиция вне границ списка.")
			return

		# Удаление узла
		if current.prev:
			current.prev.next = current.next # Устанавливаем ссылку на следующий элемент
		if current.next:
			current.next.prev = current.prev # Устанавливаем ссылку на предыдущий узел
		if current == self.head:
			self.head = current.next # Если это голова, перемещаем голову на следующий узел

		current = None # Собираем мусор, отключая узел

	def reverse(self):
		"""Обратный порядок в двусвязном списке"""
		current = self.head
		temp = None
		# Меняем местами ссылки next и prev для каждого узла
		while current:
			temp = current.prev # Храним предыдущий узел
			current.prev = current.next # Меняем next на prev
			current.next = temp # Меняем prev на  next
			current = current.prev # Переходим к следующему узлу(который теперь предшествующий)
		if temp: # Если список не пустой, обновляем голову
			self.head = temp.prev

	def search(self, value):
		"""Ищет узел по значению и возвращает его, если найден, иначе - None. """
		current = self.head
		while current:
			if current.data == value:
				return current # Возвращает узел, если значение найдено
			current = current.next
		return None # Возвращает None, если значение не найдено

	def search_return_index(self, value):
		"""Ищет узел по значению и возвращает его индекс, если найден, иначе возвращает -1"""
		current = self.head
		index = 0
		while current:
			if current.data == value:
				return index
			current = current.next
			index += 1
		return -1 # Если элемент не найден

	def print_reverse(self):
		"""Печать всех элементов списка от конца до головы."""
		current = self.head
		while current and current.next: # Найти последний узел
			current = current.next
		while current: # Печатаем в обратном порядке
			print(f"{current.data}", end = ' <-> ')
			current = current.prev
		print("None") # Для переноса строки

	def print_list(self):
		"""Печать всех элементов списка от головы до конца"""
		current = self.head
		while current:
			print(f"{current.data}", end= ' <-> ')
			current = current.next
		print("None") # Для переноса строки

class Queue:
  def __init__(self):
    self.items = []

  def enqueueFromStart(self, item):
    self.items.insert(0, item)

  def enqueueFromEnd(self, item):
    self.items.append(item)

  def isEmpty(self):
    return self.size() == 0

  def dequeueFromEnd(self):
    if not self.isEmpty():
      return self.items.pop()
    raise IndexError("dequeue from empty queue")

  def dequeueFromStart(self):
    if not self.isEmpty():
      return self.items.pop(0)
    raise IndexError("dequeue from empty queue")

  def front(self):
    if not self.isEmpty():
      return self.items[0]
    raise IndexError("front from empty queue")

  def peek(self):
    if not self.isEmpty():
      return self.items[-1]
    raise IndexError("peek from empty queue")

  def size(self):
    return len(self.items)

  def __str__(self):
    return str(self.items)

if __name__ == "__main__":
  queue = Queue()
  for i in range(10):
    queue.enqueueFromStart(i)
  print(f"Очередь: {queue}")
  print(f"Первый элемент очереди: {queue.front()}")
  print(f"Вершина очереди: {queue.peek()}")
  print(f"Удаленный элемент c конца очереди: {queue.dequeueFromEnd()}")

  print(f"Вершина очереди: {queue.peek()}")
  print(f"Удаленный элемент с начала очереди: {queue.dequeueFromStart()}")
  print(f"Очередь: {queue}")

  print(f"Первый элемент очереди: {queue.front()}")

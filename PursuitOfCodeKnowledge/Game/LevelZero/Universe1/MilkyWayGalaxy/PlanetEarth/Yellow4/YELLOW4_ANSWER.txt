import networkx as nx
import matplotlib.pyplot as plt

def graph_with_cycle():
	# Создание графа с циклом
	graph = nx.Graph()
	# Добавление узлов
	graph.add_nodes_from([1, 2, 3, 4])
	# Добавление рёбер с циклом
	graph.add_edges_from([(1, 2), (2, 3), (3, 4), (4, 1)])
	return graph

def print_graph_with_cycle(graph):
	print("Граф с циклом")
	for edge in graph.edges():
		print(f"{edge[0]} -> {edge[1]}")

def graph_with_cycle_visualization(graph):
	# Визуализация графа
	nx.draw(graph, with_labels=True, node_color='salmon', node_size=700, font_size=16)
	plt.title("Граф с циклом:")
	plt.show()

if __name__ == "__main__":
	graph = graph_with_cycle()
	print_graph_with_cycle(graph)
	graph_with_cycle_visualization(graph)

def linear_search_ai(arr, target):
  """
   Функция для линейного поиска элемента в массиве.

   :param arr: Список в котором производится поиск
   :param target: Искомый элемент
   :return: Индекс искомого элемента или -1, если элемент не найден
  """
  for index in range(len(arr)):
    if arr[index] == target:
      return index
  return -1

def linear_search_agi(arr, target):
  """
    Функция для линейного поиска элемента в массиве.

    :param arr: Список в котором производится поиск
    :param target: Искомый элемент
    :return: Индекс искомого элемента или -1, если элемент не найден
  """
  for index, value in enumerate(arr):
    if value == target:
      return index
  return -1


def main():
    my_list = [x for x in range(10)]
    print(my_list)
    try:
       my_target = int(input("Введите целое число: "))
       result = linear_search_ai(my_list, my_target)

       if result != -1:
          print(f"Элемент {my_target} найден в списке: {result}.")
       else:
          print(f"Элемент {my_target} не найден в списке.")
    except ValueError:
          print("Пожалуйста введите корректное целое число")


    my_list_search_agi = my_list.copy()
    try:
        my_target = int(input("Введите целое число: "))
        result = linear_search_agi(my_list_search_agi, my_target)

        if result != -1:
            print(f"Элемент {my_target} найден в списке: {result}")
        else:
            print(f"Элемент {my_target} не найден в списке")
    except ValueError:
        print("Пожалуйста введите корректное целое число")

if __name__ == "__main__":
    main()

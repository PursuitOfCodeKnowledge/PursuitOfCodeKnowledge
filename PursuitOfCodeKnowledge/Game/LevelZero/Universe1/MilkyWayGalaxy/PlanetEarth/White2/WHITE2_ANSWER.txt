pwd
К примеру она тебе покажет путь /home/sergey/Desktop/PursuitOfCodeKnowledge.git/PursuitOfCodeKnowledge/Game/LevelZero/Universe1/MilkyWayGalaxy/PlanetEarth/White2, получается ты сейчас находишься в White2,

ls
или
ls -a

Пишем в файле:
from collections import deque
if __name__ == "__main__":
  queue = deque()
  for i in range(10):
    queue.append(i)

  print(f"Очередь: {queue}")
  print(f"Вершина очереди: {queue[-1]}")
  print(f"Первый элемент очереди: {queue[0]}")
  print(f"Удаленный элемент: {queue.popleft()}")
  print(f"Очередь: {queue}")
  print(f"Вершина очереди: {queue[-1]}")
  print(f"Первый элемент очереди: {queue[0]}")
  print(f"Вставка вначала очереди числа 9:")
  queue.insert(0,9)
  print(f"Очередь: {queue}")
  print(f"Вершина очереди: {queue[-1]}")
  print(f"Первый элемент очереди: {queue[0]}")
  print(f"Удаленный элемент с конца: {queue.pop()}")
  print(f"Очередь: {queue}")
  print(f"Вершина очереди: {queue[-1]}")
  print(f"Первый элемент очереди: {queue[0]}")

import networkx as nx
import matplotlib.pyplot as plt

def create_graph():
	# Создание графа
	graph = nx.Graph()
	# Добавление рёбер с весами
	graph.add_weighted_edges_from([(1, 2, 1), (1, 3, 4), (2, 3, 2), (2, 4, 5), (3, 4, 1)])
	return graph

def find_shortest_path_in_graph(graph):
	# Поиск кратчайшего пути
	path = nx.shortest_path(graph, source=1, target=4, weight='weight')
	return path

def print_graph(graph):
	# Вывод графа в терминал
	print("Граф с весами:")
	for (u, v, weight) in graph.edges(data='weight'):
		print(f"Узел {u} -- Узел {v} (Вес: {weight})")

def graph_visualization(graph):
	# Визуализация графа
	pos = nx.spring_layout(graph)
	nx.draw(graph, pos, with_labels=True, node_color='lightyellow', node_size=700, font_size=16)
	nx.draw_networkx_edge_labels(graph, pos, edge_labels={(1, 2): 1, (1, 3): 4, (2, 3): 2, (2, 4): 5, (3, 4): 1})
	plt.title("Взвешенный граф:")
	plt.show()

if __name__ == "__main__":
	graph = create_graph()
	print_graph(graph)
	print("Кратчайший путь от 1 до 4: ", find_shortest_path_in_graph(graph))
	graph_visualization(graph)
